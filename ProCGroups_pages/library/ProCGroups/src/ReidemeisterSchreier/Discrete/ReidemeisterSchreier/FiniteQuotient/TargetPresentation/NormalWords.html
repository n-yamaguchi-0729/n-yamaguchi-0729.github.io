26/--
27A computation-facing variant of \(\mathrm{CleanedReidemeisterSchreierTargetPresentationData}\),
28carrying a concrete target word and a certificate that substitution into the reduced
29\(\mathrm{cleanedTauNormalWord}(q,r)\) reduces to that word.
30-/
31structure CleanedReidemeisterSchreierTargetPresentationNormalWordData
32 (R : Set (FreeGroup X)) (Y : Type*)
33 [DecidablePred D.toFiniteQuotientSchreierData.IsDegenerateSchreierSymbol]
34 [DecidableEq D.nondegenerateSchreierSymbol] [DecidableEq Y] where
35 /-- The relators of the chosen target presentation on generators \(Y\). -/
36 targetRelators : Set (FreeGroup Y)
37 /-- The word in target generators assigned to each nondegenerate Schreier generator. -/
38 toTargetGenerator : D.nondegenerateSchreierSymbol → FreeGroup Y
39 /-- The word in nondegenerate Schreier generators assigned to each target generator. -/
40 fromTargetGenerator : Y → FreeGroup D.nondegenerateSchreierSymbol
41 /-- A concrete target word representing each cleaned Reidemeister--Schreier relator. -/
42 cleanedRelatorWord : Q → FreeGroup X → List (Y × Bool)
43 /--
44 Substituting target words into a cleaned normal word reduces to the specified target relator
45 word.
46 -/
47 cleanedRelatorWord_reduce :
48 ∀ q : Q, ∀ r ∈ R,
49 FreeGroup.reduce
50 (freeGroupSubstitutionWord toTargetGenerator
51 (D.cleanedTauNormalWord q r)) =
52 FreeGroup.reduce (cleanedRelatorWord q r)
53 /-- Every specified target relator word is trivial modulo the target relators. -/
54 mapsCleanedRelatorWords :
55 ∀ q : Q, ∀ r ∈ R,
56 RelatorEquivalent targetRelators
57 (FreeGroup.mk (cleanedRelatorWord q r)) 1
58 /-- Target relators map to consequences of the cleaned Reidemeister--Schreier relators. -/
59 mapsTargetRelators :
60 ∀ s ∈ targetRelators,
61 RelatorEquivalent (D.cleanedSchreierRelators R)
62 (FreeGroup.lift fromTargetGenerator s) 1
63 /-- Mapping a nondegenerate Schreier generator forward and back recovers it modulo relators. -/
64 from_toTargetGenerator :
65 ∀ z : D.nondegenerateSchreierSymbol,
66 RelatorEquivalent (D.cleanedSchreierRelators R)
67 (FreeGroup.lift fromTargetGenerator (toTargetGenerator z))
68 (FreeGroup.of z)
69 /-- Mapping a target generator back and forward recovers it modulo target relators. -/
70 to_fromTargetGenerator :
71 ∀ y : Y,
72 RelatorEquivalent targetRelators
73 (FreeGroup.lift toTargetGenerator (fromTargetGenerator y))
74 (FreeGroup.of y)

83/--
84The normal-word target data maps every cleaned Schreier relator to an identity modulo the target
85relators.
86-/
87theorem mapsCleanedRelators
88 (C : D.CleanedReidemeisterSchreierTargetPresentationNormalWordData R Y) :
89 ∀ q : Q, ∀ r ∈ R,
90 RelatorEquivalent C.targetRelators
91 (FreeGroup.lift C.toTargetGenerator (D.cleanedTau q r)) 1 :=
92 D.toFiniteQuotientSchreierData.mapsCleanedRelators_of_cleanedTauNormalWord_reduce
93 (toTargetGenerator := C.toTargetGenerator)
94 C.cleanedRelatorWord C.cleanedRelatorWord_reduce
95 C.mapsCleanedRelatorWords

97/--
98Normal-word target presentation data forgets to the corresponding cleaned target presentation
99data.
100-/
101def toTargetPresentationData
102 (C : D.CleanedReidemeisterSchreierTargetPresentationNormalWordData R Y) :
103 D.CleanedReidemeisterSchreierTargetPresentationData R Y where
104 targetRelators := C.targetRelators
105 toTargetGenerator := C.toTargetGenerator
106 fromTargetGenerator := C.fromTargetGenerator
107 mapsCleanedRelators := C.mapsCleanedRelators
108 mapsTargetRelators := C.mapsTargetRelators
109 from_toTargetGenerator := C.from_toTargetGenerator
110 to_fromTargetGenerator := C.to_fromTargetGenerator

112/-- The normal-word cleaned target presentation data yields a Tietze equivalence. -/
113def toTietzeEquiv

115 TietzeEquiv (D.cleanedSchreierRelators R) C.targetRelators :=

119The normal-word target presentation is equivalent to the cleaned Reidemeister--Schreier

128/--
129The target presented group is equivalent to the subgroup kernel described by the cleaned
130Reidemeister--Schreier presentation.
131-/
132noncomputable def targetPresentedGroupEquivKernel

134 (hR : Subgroup.normalClosure R ≤ D.kernel) :
135 PresentedGroup C.targetRelators ≃*
