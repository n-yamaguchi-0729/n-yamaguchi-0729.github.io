23/--
24This is the relator family obtained by rewriting original relators and then deleting degenerate
25Schreier generators.
26-/
27def cleanedSchreierRelators
28 (R : Set (FreeGroup X))
29 [DecidablePred D.IsDegenerateSchreierSymbol] :
30 Set (FreeGroup D.NondegenerateSchreierSymbol) :=
31 { z | ∃ q : Q, ∃ r ∈ R, z = D.cleanedTau q r }
32
33/--
34Membership in the cleaned Schreier relator set is characterized by a quotient element and an
35original relator whose cleaned rewrite gives the word.
36-/
37theorem mem_cleanedSchreierRelators
38 (R : Set (FreeGroup X))
39 [DecidablePred D.IsDegenerateSchreierSymbol]
40 (q : Q) {r : FreeGroup X} (hr : r ∈ R) :
41 D.cleanedTau q r ∈ D.cleanedSchreierRelators R :=
42 ⟨q, r, hr, rfl⟩
43
44/--
45The cleaned Schreier relator finset contains the cleaned rewrites of the finite quotient
46elements and original relators.
47-/
48noncomputable def cleanedSchreierRelatorFinset
49 [DecidablePred D.IsDegenerateSchreierSymbol]
50 [DecidableEq (FreeGroup D.NondegenerateSchreierSymbol)]
51 (R : Finset (FreeGroup X)) :
52 Finset (FreeGroup D.NondegenerateSchreierSymbol) :=
53 (Finset.univ : Finset Q).biUnion
54 (fun q => R.image (fun r => D.cleanedTau q r))
55
56/--
57Membership in the cleaned Schreier relator finset is characterized by a quotient element and an
58original relator whose cleaned rewrite gives the word.
59-/
60@[simp]
61theorem mem_cleanedSchreierRelatorFinset
62 {D : FiniteQuotientSchreierData X Q}
63 [DecidablePred D.IsDegenerateSchreierSymbol]
64 [DecidableEq (FreeGroup D.NondegenerateSchreierSymbol)]
65 {R : Finset (FreeGroup X)} {z : FreeGroup D.NondegenerateSchreierSymbol} :
66 z ∈ D.cleanedSchreierRelatorFinset R ↔
67 z ∈ D.cleanedSchreierRelators ({r | r ∈ R} : Set (FreeGroup X)) := by
68 simp only [cleanedSchreierRelatorFinset, Finset.mem_biUnion, Finset.mem_univ,
69 Finset.mem_image, eq_comm,
70 true_and, cleanedSchreierRelators, SetLike.setOf_mem_eq, SetLike.mem_coe, Set.mem_setOf_eq]

72/-- Coercing the cleaned Schreier relator finset gives the cleaned Schreier relator set. -/
73@[simp]
74theorem coe_cleanedSchreierRelatorFinset
75 [DecidablePred D.IsDegenerateSchreierSymbol]
76 [DecidableEq (FreeGroup D.NondegenerateSchreierSymbol)]
77 (R : Finset (FreeGroup X)) :
78 ({z | z ∈ D.cleanedSchreierRelatorFinset R} :
79 Set (FreeGroup D.NondegenerateSchreierSymbol)) =
80 D.cleanedSchreierRelators ({r | r ∈ R} : Set (FreeGroup X)) := by
81 ext z
82 simp only [mem_cleanedSchreierRelatorFinset, SetLike.setOf_mem_eq, Set.setOf_mem_eq]

84/--
85The relators after deleting degenerate Schreier generators are exactly the cleaned Schreier
86relators.
87-/
88theorem presentationRelatorsAfterDeletingDegenerate_eq_cleaned
89 (R : Set (FreeGroup X))
90 [DecidablePred D.IsDegenerateSchreierSymbol] :
91 D.presentationRelatorsAfterDeletingDegenerateSchreierGenerators R =
92 D.cleanedSchreierRelators R := by
93 ext z
94 change z ∈
95 Presented.trivializeGeneratorsHom
96 D.NondegenerateSchreierSymbol D.DegenerateSchreierSymbol ''
97 (FreeGroup.freeGroupCongr
98 (Presented.GeneratorPartition.equiv D.IsDegenerateSchreierSymbol) ''
99 D.schreierRelators R) ↔
100 z ∈ D.cleanedSchreierRelators R
101 constructor
102 · rintro ⟨y, ⟨x, hx, rfl⟩, rfl⟩
103 rcases hx with ⟨q, r, hr, rfl⟩
104 exact ⟨q, r, hr, rfl⟩
105 · rintro ⟨q, r, hr, rfl⟩
106 exact
107 ⟨FreeGroup.freeGroupCongr
108 (Presented.GeneratorPartition.equiv D.IsDegenerateSchreierSymbol)
109 (D.tau q r),
110 ⟨D.tau q r, ⟨q, r, hr, rfl⟩, rfl⟩, rfl⟩

112/--
113Relators forcing the chosen quotient representatives to rewrite to the empty word. These are
114automatic for a prefix-closed Schreier transversal, but are useful for arbitrary finite quotient
115sections.
116-/
117def quotientSectionRelators :
118 Set (FreeGroup (FiniteSchreierSymbol X Q)) :=
119 { q | ∃ s : Q, q = D.tau 1 (D.quotientSection s) }
120
121/--
122The augmented presentation relators combine quotient-section relators with the presentation
123relators.
124-/
125def augmentedPresentationRelators (R : Set (FreeGroup X)) :
126 Set (FreeGroup (FiniteSchreierSymbol X Q)) :=
127 D.presentationRelators R ∪ D.quotientSectionRelators

129/-- A finite set enumerating the finite-quotient Schreier relators. -/
130noncomputable def schreierRelatorFinset
131 (R : Finset (FreeGroup X)) :
132 Finset (FreeGroup (FiniteSchreierSymbol X Q)) := by
133 classical
134 exact Finset.univ.biUnion (fun s : Q => R.image (D.tau s))

136/--
137Membership in the Schreier-relator finset is equivalent to membership in the finite-quotient
138Schreier relator set.
139-/
140theorem mem_schreierRelatorFinset
141 {D : FiniteQuotientSchreierData X Q}
142 {R : Finset (FreeGroup X)}
143 {z : FreeGroup (FiniteSchreierSymbol X Q)} :
144 z ∈ D.schreierRelatorFinset R ↔
145 z ∈ D.schreierRelators ({r | r ∈ R} : Set (FreeGroup X)) := by
146 classical
147 simp only [schreierRelatorFinset, Finset.mem_biUnion, Finset.mem_univ, Finset.mem_image,
148 eq_comm, true_and,
149 schreierRelators, SetLike.setOf_mem_eq, SetLike.mem_coe, Set.mem_setOf_eq]

151/-- A finite set enumerating the degenerate finite-quotient Schreier relators. -/
152noncomputable def degenerateSchreierRelatorFinset [Fintype X] :
153 Finset (FreeGroup (FiniteSchreierSymbol X Q)) := by
154 classical
155 exact (((Finset.univ : Finset Q).product (Finset.univ : Finset X)).filter
156 (fun z =>
157 D.quotientSection (D.transition z.1 z.2) =
158 D.quotientSection z.1 * FreeGroup.of z.2)).image
159 (fun z => FreeGroup.of z)

161/--
162Membership in the degenerate-Schreier-relator finset is equivalent to membership in the
163degenerate Schreier relator set.
164-/
165theorem mem_degenerateSchreierRelatorFinset [Fintype X]
166 {D : FiniteQuotientSchreierData X Q}
167 {z : FreeGroup (FiniteSchreierSymbol X Q)} :
168 z ∈ D.degenerateSchreierRelatorFinset ↔
169 z ∈ D.degenerateSchreierRelators := by
170 classical
171 simp only [degenerateSchreierRelatorFinset, transition_eq, eq_comm, Finset.product_eq_sprod,
172 Finset.univ_product_univ, Finset.mem_image, Finset.mem_filter, Finset.mem_univ, true_and,
173 Prod.exists,
174 degenerateSchreierRelators, Set.mem_setOf_eq]

176/-- A finite set enumerating the finite-quotient presentation relators. -/
177noncomputable def presentationRelatorFinset [Fintype X]
178 (R : Finset (FreeGroup X)) :
179 Finset (FreeGroup (FiniteSchreierSymbol X Q)) := by
180 classical
181 exact D.schreierRelatorFinset R ∪ D.degenerateSchreierRelatorFinset

183/--
184Membership in the presentation-relator finset is equivalent to membership in the finite-quotient
185presentation relator set.
186-/
187theorem mem_presentationRelatorFinset [Fintype X]
188 {D : FiniteQuotientSchreierData X Q}
189 {R : Finset (FreeGroup X)}
190 {z : FreeGroup (FiniteSchreierSymbol X Q)} :
191 z ∈ D.presentationRelatorFinset R ↔
192 z ∈ D.presentationRelators ({r | r ∈ R} : Set (FreeGroup X)) := by
193 classical
194 simp only [presentationRelatorFinset, Finset.mem_union, mem_schreierRelatorFinset,
195 SetLike.setOf_mem_eq,
196 mem_degenerateSchreierRelatorFinset, presentationRelators, Set.mem_union]

198/-- The presentation-relator finset coerces to the finite-quotient presentation relator set. -/
199theorem coe_presentationRelatorFinset [Fintype X]
200 (R : Finset (FreeGroup X)) :
201 ({z | z ∈ D.presentationRelatorFinset R} :
202 Set (FreeGroup (FiniteSchreierSymbol X Q))) =
203 D.presentationRelators ({r | r ∈ R} : Set (FreeGroup X)) := by
204 ext z
205 exact D.mem_presentationRelatorFinset (R := R) (z := z)
