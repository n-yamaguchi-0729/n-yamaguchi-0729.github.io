34/-- Common word-level data for a finite quotient section. -/
35structure QuotientSectionWordCertificate
36 (X Q : Type*) [Group Q] [Fintype Q] where
37 /-- The homomorphism from the free group to the finite quotient. -/
38 quotientMap : FreeGroup X →* Q
39 /-- The chosen word representing each finite quotient element. -/
40 quotientSectionWord : Q → List (X × Bool)
41 /-- Evaluating a chosen word and applying the quotient map recovers its quotient element. -/
42 quotientMap_mk_quotientSectionWord :
43 ∀ q : Q, quotientMap (FreeGroup.mk (quotientSectionWord q)) = q

45/--
46A reusable word-level certificate for a prefix-closed finite quotient section, including the
47required reduced-word and prefix-closedness conditions.
48-/
49structure PrefixClosedQuotientSectionWordCertificate
50 (X Q : Type*) [Group Q] [Fintype Q] [DecidableEq X]
51 extends QuotientSectionWordCertificate X Q where
52 /-- The identity quotient element is represented by the empty word. -/
53 quotientSectionWord_one : quotientSectionWord 1 = []
54 /-- Every chosen quotient-section word is reduced. -/
55 quotientSectionWord_reduced :
56 ∀ q : Q, FreeGroup.reduce (quotientSectionWord q) = quotientSectionWord q
57 /-- Every prefix of a chosen quotient-section word is the chosen word for its quotient state. -/
58 prefixClosed_quotientSectionWord :
59 ∀ q : Q,
60 (FiniteQuotientSchreierData.ofQuotientSectionWords quotientMap
61 quotientSectionWord quotientMap_mk_quotientSectionWord
62 quotientSectionWord_one).prefixClosedQuotientSectionWordAlongList
63 quotientSectionWord 1 (quotientSectionWord q)

92/--
93A certificate for a positive prefix-closed quotient-section tree. Each quotient element has a
94positive word in the original generators, and every prefix is the chosen word for the quotient
95element it represents.
96-/
97structure PositivePrefixClosedQuotientSectionWordCertificate
98 (X Q : Type*) [Group Q] [Fintype Q]
99 extends QuotientSectionWordCertificate X Q where
100 /-- Every letter in every chosen quotient-section word has positive orientation. -/
101 quotientSectionWord_positive :
102 ∀ q : Q, ∀ xb ∈ quotientSectionWord q, xb.2 = true
103 /-- Each prefix is the chosen word for the quotient element obtained by evaluating that prefix. -/
104 prefixState :
105 ∀ q : Q, ∀ acc rest : List (X × Bool),
106 acc ++ rest = quotientSectionWord q →
107 quotientSectionWord (quotientMap (FreeGroup.mk acc)) = acc

113/-- The quotient-section word certificate sends the identity quotient element to the empty word. -/
114theorem quotientSectionWord_one :
115 C.quotientSectionWord 1 = [] := by
116 have h := C.prefixState 1 [] (C.quotientSectionWord 1) (by simp only [List.nil_append])
117 simpa [← FreeGroup.one_eq_mk] using h
118
119/-- Every quotient-section word in the positive prefix-closed certificate is reduced. -/
120theorem quotientSectionWord_reduced [DecidableEq X] (q : Q) :
121 FreeGroup.reduce (C.quotientSectionWord q) = C.quotientSectionWord q :=
122 FiniteQuotientSchreierData.reduce_eq_of_forall_snd_eq_true
123 (C.quotientSectionWord_positive q)

131 C.quotientSectionWord_one

135 PrefixClosedFiniteQuotientSchreierData X Q where
136 toFiniteQuotientSchreierData := C.toFiniteQuotientSchreierData
137 isPrefixClosed := by
138 apply
139 FiniteQuotientSchreierData.isPrefixClosedQuotientSection_of_prefixClosedQuotientSectionWord
140 (D := C.toFiniteQuotientSchreierData)
141 · intro q
142 rfl
143 · intro q
144 exact C.quotientSectionWord_reduced q
145 · intro q
146 exact
147 FiniteQuotientSchreierData.prefixClosedQuotientSectionWordAlongList_of_positive_prefixStates
148 (D := C.toFiniteQuotientSchreierData)
149 (by intro q; rfl)
150 C.quotientSectionWord_positive
151 (by
152 intro target acc rest hcat
153 simpa [toFiniteQuotientSchreierData,
154 FiniteQuotientSchreierData.ofQuotientSectionWords] using
155 C.prefixState target acc rest hcat)
156 q

438/--
439A certificate that the cleaned finite Reidemeister--Schreier presentation is Tietze equivalent
440to a chosen target presentation.
441-/
442structure CleanedReidemeisterSchreierTargetPresentationData
443 (R : Set (FreeGroup X)) (Y : Type*)
444 [DecidablePred D.toFiniteQuotientSchreierData.IsDegenerateSchreierSymbol] where
445 /-- The relators of the chosen target presentation on generators \(Y\). -/
446 targetRelators : Set (FreeGroup Y)
447 /-- The word in target generators assigned to each nondegenerate Schreier generator. -/
448 toTargetGenerator : D.nondegenerateSchreierSymbol → FreeGroup Y
449 /-- The word in nondegenerate Schreier generators assigned to each target generator. -/
450 fromTargetGenerator : Y → FreeGroup D.nondegenerateSchreierSymbol
451 /-- Cleaned Reidemeister--Schreier relators map to consequences of the target relators. -/
452 mapsCleanedRelators :
453 ∀ q : Q, ∀ r ∈ R,
454 RelatorEquivalent targetRelators
455 (FreeGroup.lift toTargetGenerator (D.cleanedTau q r)) 1
456 /-- Target relators map to consequences of the cleaned Reidemeister--Schreier relators. -/
457 mapsTargetRelators :
458 ∀ s ∈ targetRelators,
459 RelatorEquivalent (D.cleanedSchreierRelators R)
460 (FreeGroup.lift fromTargetGenerator s) 1
461 /-- Mapping a nondegenerate Schreier generator forward and back recovers it modulo relators. -/
462 from_toTargetGenerator :
463 ∀ z : D.nondegenerateSchreierSymbol,
464 RelatorEquivalent (D.cleanedSchreierRelators R)
465 (FreeGroup.lift fromTargetGenerator (toTargetGenerator z))
466 (FreeGroup.of z)
467 /-- Mapping a target generator back and forward recovers it modulo target relators. -/
468 to_fromTargetGenerator :
469 ∀ y : Y,
470 RelatorEquivalent targetRelators
471 (FreeGroup.lift toTargetGenerator (fromTargetGenerator y))
472 (FreeGroup.of y)
