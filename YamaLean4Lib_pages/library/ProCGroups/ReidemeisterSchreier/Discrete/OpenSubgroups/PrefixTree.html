  exact fun a b =>
    { e |
        ∃ hw : FreeGroup.toWord (((show ActionCategory (FreeGroup X) T from b).back : T) :
            FreeGroup X) ≠ [],

            prefixParent_mem_of_mem (X := X) hT tb.property⟩
